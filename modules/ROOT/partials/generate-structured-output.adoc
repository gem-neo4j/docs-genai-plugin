# tag::classification[]
MATCH (n:Movie)
WHERE n.imdbRating IS NOT NULL
ORDER BY n.imdbRating DESC
LIMIT 50
WITH n.title + ': ' + n.plot AS movie
WITH
  reduce(acc = '', item IN collect(movie) | acc + item + '\n') AS movies,
  {
    type: 'object',
    properties: {
      movies: {
        type: 'array',
        items: {
          type: 'object',
          properties: {
            genre: {type: "string"},
            movieTitles: {type: "array", items: {type: 'string'}}
          },
          required: ["genre", "movieTitles"],
          additionalProperties: false
        }
      }
    },
    required: ['movies'],
    additionalProperties: false
  } AS schema,
# end::classification[]
